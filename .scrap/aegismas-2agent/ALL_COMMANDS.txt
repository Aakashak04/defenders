================================================================================
1. BUILD THE AGENT
================================================================================

cd aegismas-2agent
docker compose build

Status: ✅ Already built!

================================================================================
2. TRAIN MODELS
================================================================================

# Generate training data (DONE ✅)
python scripts/gen_synthetic.py

# Train LSTM model (DONE ✅)
python scripts/train_lstm.py --epochs 5

# Optional: Train statistical model (runs in bootstrap mode if skipped)
# See QUICKSTART.md for Python script

================================================================================
3. RUN THE DEFENDER
================================================================================

# Start all services (Envoy + AegisMAS API + Redis)
docker compose up -d

# View logs
docker compose logs -f

# Stop defender
docker compose down

# Restart
docker compose restart

================================================================================
4. RUN THE ATTACKER
================================================================================

# Test WITHOUT defender (direct to bank at localhost:8000)
python attack_simulator.py --defender off

# Test WITH defender (through Envoy gateway at localhost:8085)
python attack_simulator.py --defender on

# Quiet mode (summary only)
python attack_simulator.py --defender on --quiet

================================================================================
5. VIEW RESULTS
================================================================================

# Health check
curl http://localhost:8080/health

# View all events
curl http://localhost:8080/v1/events

# View only blocked events
curl "http://localhost:8080/v1/events?decision=block"

# View metrics
curl http://localhost:8080/metrics

# PowerShell alternative
Invoke-WebRequest -Uri "http://localhost:8080/v1/events?decision=block" | Select-Object -ExpandProperty Content | ConvertFrom-Json | ConvertTo-Json -Depth 10

================================================================================
6. COMPLETE WORKFLOW
================================================================================

# Navigate to project
cd aegismas-2agent

# Start defender
docker compose up -d

# Wait for services (10 seconds)
Start-Sleep -Seconds 10

# Test with defender ON
python attack_simulator.py --defender on

# View blocked events
curl "http://localhost:8080/v1/events?decision=block"

# View metrics
curl http://localhost:8080/metrics

# Stop when done
docker compose down

================================================================================
7. TROUBLESHOOTING
================================================================================

# Check services status
docker compose ps

# View logs
docker compose logs aegismas-api
docker compose logs envoy

# Rebuild and restart
docker compose up --build -d

# Check bank is running
curl http://localhost:8000/health

# Test Envoy directly
curl http://localhost:8085/health
